cmake_minimum_required(VERSION 3.20)
project(AetherOS VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(aether_core src/AetherCore.cpp)
target_include_directories(aether_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(aether_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(aether_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(aether src/main.cpp)
target_link_libraries(aether PRIVATE aether_core)
enable_testing()
add_executable(aether_tests tests/CoreTests.cpp)
target_link_libraries(aether_tests PRIVATE aether_core)
add_test(NAME aether_core_tests COMMAND aether_tests)

if(WIN32)
  add_executable(AetherOS WIN32 src/WindowsMainframe.cpp)
  target_link_libraries(AetherOS PRIVATE aether_core comdlg32)
  target_compile_definitions(AetherOS PRIVATE UNICODE _UNICODE)
endif()
